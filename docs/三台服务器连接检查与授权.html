# Run as superopen on 192.168.10.101; append only, preserve existing keys.
set -eu
if [ "$(id -un)" != "superopen" ]; then printf "Please run as superopen, without sudo.\n" >&2; exit 1; fi
umask 077
mkdir -p "$HOME/.ssh"
chmod 700 "$HOME/.ssh"
key='ssh-ed25519 AAAAC3NzaC1lZDI1NTE5AAAAIBNUFmQyCelq3IrrCmfMfFZwDVciVaNeKAOG6ikZrdpw jarvis-lenovo-server-101'
auth="$HOME/.ssh/authorized_keys"
if [ ! -f "$auth" ]; then : > "$auth"; fi
# Ensure a separator if an existing final line has no newline.
if ! grep -qxF "$key" "$auth"; then printf '\n%s\n' "$key" >> "$auth"; fi
chmod 600 "$auth"
printf 'Jarvis 101 public key authorized for %s\n' "$(id -un)"
